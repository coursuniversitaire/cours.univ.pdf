const HTML_PROPERTY_KEY = "HTML_STR_LAST_HANDLED_INDEX";
const SITE_BASE_URL = "https://coursuniversitaire.github.io/cours.univ.pdf/";

// ---------- NAMING / URL HELPERS (single source of truth) ----------
// Everything that needs a module's file name or public URL must go through
// these two functions. Never rebuild "exercices-corriges-...-pdf.html" inline.

function getModuleFileName(folderName) {
  return "exercices-corriges-" + slugify(folderName) + "-pdf.html";
}

function getModulePageUrl(folderName) {
  return SITE_BASE_URL + "matieres/" + getModuleFileName(folderName);
}

// ---------- MODULE PAGES ----------

function generateAllModulePages() {
  const parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
  const props = PropertiesService.getScriptProperties();

  let webFolder;
  const webIter = parentFolder.getFoldersByName(WEB_FOLDER_NAME);
  webFolder = webIter.hasNext() ? webIter.next() : parentFolder.createFolder(WEB_FOLDER_NAME);

  const subFolders = [];
  const iterator = parentFolder.getFolders();
  while (iterator.hasNext()) {
    const f = iterator.next();
    if (f.getName() === WEB_FOLDER_NAME) continue;
    subFolders.push({ name: f.getName(), id: f.getId(), ud: f.getLastUpdated() });
  }
  subFolders.sort((a, b) => b.ud - a.ud);

  const startIndex = parseInt(props.getProperty(HTML_PROPERTY_KEY) || "0", 10);
  Logger.log(`[HTML] Resuming from index ${startIndex} / ${subFolders.length}`);

  let count = 0, summary = "";

  for (let i = startIndex; i < subFolders.length; i++) {
    const { name: folderName, id: folderId } = subFolders[i];
    const subFolder = DriveApp.getFolderById(folderId);

    Logger.log(`[${i + 1}/${subFolders.length}] Generating HTML: ${folderName}`);

    try {
      const htmlFileName = getModuleFileName(folderName);
      const existingFiles = webFolder.getFilesByName(htmlFileName);
      while (existingFiles.hasNext()) { existingFiles.next().setTrashed(true); }

      const visited = new Set();
      const treeText = getFolderTree(subFolder, "", visited);
      const stats = countDriveItems(subFolder);

      const html = buildModulePage(folderName, treeText, stats);
      webFolder.createFile(htmlFileName, html, MimeType.HTML);

      summary += `  [${i + 1}] ${folderName} → ${htmlFileName}\n`;
    } catch (e) {
      summary += `  [${i + 1}] ${folderName} → ERROR: ${e.message}\n`;
      Logger.log(`Error on "${folderName}": ${e.message}`);
    }

    props.setProperty(HTML_PROPERTY_KEY, String(i + 1));
    count++;
  }

  Logger.log(`Done! Generated ${count} HTML pages:\n${summary}`);
}

function resetHtmlProgress() {
  PropertiesService.getScriptProperties().deleteProperty(HTML_PROPERTY_KEY);
  Logger.log("HTML progress reset. Next run starts from index 0.");
}

function countDriveItems(folder) {
  let cours = 0, td = 0, examens = 0, tp = 0, autres = 0;

  function walk(f) {
    const files = f.getFiles();
    while (files.hasNext()) {
      const file = files.next();
      const n = file.getName().toLowerCase();
      if (n.includes("cours")) cours++;
      else if (n.includes("td") || n.includes("exercice")) td++;
      else if (n.includes("exam") || n.includes("contr")) examens++;
      else if (n.includes("tp")) tp++;
      else autres++;
    }
    const subs = f.getFolders();
    while (subs.hasNext()) walk(subs.next());
  }

  walk(folder);
  return { cours, td, examens, tp, autres };
}

function slugify(name) {
  return name
    .toLowerCase()
    .replace(/[àáâãäå]/g, "a")
    .replace(/[éèêë]/g, "e")
    .replace(/[îï]/g, "i")
    .replace(/[ôö]/g, "o")
    .replace(/[ùûü]/g, "u")
    .replace(/[ç]/g, "c")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

// Converts plain-text tree (with box-drawing chars) to styled HTML
function treeToHtml(treeText) {
  const lines = treeText.split("\n");
  let html = "";
  lines.forEach(line => {
    if (!line.trim()) return;
    const escaped = line
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;");

    let cls = "tree-file";
    if (line.includes("📁")) cls = "tree-folder";
    else if (line.includes("🔗📁")) cls = "tree-shortcut";

    html += `<div class="${cls}">${escaped}</div>\n`;
  });
  return html;
}

function generateHomePage() {
  const parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);

  let webFolder;
  const webIter = parentFolder.getFoldersByName(WEB_FOLDER_NAME);
  webFolder = webIter.hasNext() ? webIter.next() : parentFolder.createFolder(WEB_FOLDER_NAME);

  const subFolders = [];
  const iterator = parentFolder.getFolders();
  while (iterator.hasNext()) {
    const f = iterator.next();
    if (f.getName() === WEB_FOLDER_NAME) continue;
    subFolders.push({ name: f.getName(), id: f.getId() });
  }
  subFolders.sort((a, b) => a.name.localeCompare(b.name, "fr"));

  const html = buildHomePage(subFolders);

  const existing = webFolder.getFilesByName(HOME_FILE_NAME);
  while (existing.hasNext()) { existing.next().setTrashed(true); }
  webFolder.createFile(HOME_FILE_NAME, html, MimeType.HTML);

  Logger.log(`Home page generated with ${subFolders.length} module links → ${HOME_FILE_NAME}`);
  Logger.log(`Upload ${WEB_FOLDER_NAME}/${HOME_FILE_NAME} to the ROOT of your GitHub Pages repo as index.html`);
}
